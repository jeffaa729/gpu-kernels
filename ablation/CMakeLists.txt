cmake_minimum_required(VERSION 3.24)
project(gemm_ablation LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(GPU_KERNELS_ABLATION_ARCH "89" CACHE STRING "Ablation architecture: 89 or 90a")
set(CMAKE_CUDA_ARCHITECTURES ${GPU_KERNELS_ABLATION_ARCH})
find_package(CUDAToolkit REQUIRED)
if(GPU_KERNELS_ABLATION_ARCH STREQUAL "90a")
    file(GLOB STAGES CONFIGURE_DEPENDS sm90_bf16/*.cu)
else()
    file(GLOB STAGES CONFIGURE_DEPENDS sm89_fp32/*.cu sm89_bf16/*.cu)
endif()
add_library(gemm_ablation SHARED bridge.cu ${STAGES})
if(GPU_KERNELS_ABLATION_ARCH STREQUAL "90a")
    target_compile_definitions(gemm_ablation PRIVATE ABLATION_SM90)
endif()
target_link_libraries(gemm_ablation PRIVATE CUDA::cudart CUDA::cublas CUDA::cuda_driver)
target_compile_options(gemm_ablation PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>)
add_executable(profile_gemm profile_gemm.cu)
target_link_libraries(profile_gemm PRIVATE gemm_ablation CUDA::cudart)
