cmake_minimum_required(VERSION 3.24)
project(gemm_ablation LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 89)
find_package(CUDAToolkit REQUIRED)
file(GLOB STAGES CONFIGURE_DEPENDS sm89_fp32/*.cu sm89_bf16/*.cu)
add_library(gemm_ablation SHARED bridge.cu ${STAGES})
target_link_libraries(gemm_ablation PRIVATE CUDA::cudart CUDA::cublas)
target_compile_options(gemm_ablation PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>)
add_executable(profile_gemm profile_gemm.cu)
target_link_libraries(profile_gemm PRIVATE gemm_ablation CUDA::cudart)
